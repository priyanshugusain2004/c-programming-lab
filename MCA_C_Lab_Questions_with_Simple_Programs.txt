    for (; temp > 0; temp = temp / 10)
    {
        digit = temp % 10;

            fact = fact * i;
 
        sum = sum + fact;

        sum = 1;

        printf("Strong Number");
    else
    {
        printf("Not a Strong Number\\n");

        {
            temp = num;
            sum = 0;
 
            for (; temp > 0; temp = temp / 10)
            {
                digit = temp % 10;
                fact = 1;
 
                for (i = 1; i <= digit; i++)
                    fact = fact * i;
 
                sum = sum + fact;
            }
 
            if (num == 0)
                sum = 1;
 
            if (sum == num)
            {

13. Write a program using functions to check whether a number is an Armstrong number (call by value).
Program:
#include <stdio.h>
 
int power(int base, int exp)
{
    int result = 1, i;
 
    for (i = 1; i <= exp; i++)
        result = result * base;
 
    return result;
}
 
int isArmstrong(int n)
{
    int temp = n, digits = 0, digit, sum = 0;
 
    if (n == 0)
        digits = 1;
 
    for (temp = n; temp > 0; temp = temp / 10)
        digits++;
 
    temp = n;
 
    for (; temp > 0; temp = temp / 10)
    {
        digit = temp % 10;
        sum = sum + power(digit, digits);
    }
 
    return sum == n;
}
 
int main()
{
    int n;
 
    printf("Enter a number: ");
    scanf("%d", &n);
 
    if (n >= 0 && isArmstrong(n))
        printf("Armstrong Number");
    else
        printf("Not an Armstrong Number");
 
    return 0;
}
14. Write a program to demonstrate call by value vs call by address using two functions that attempt to swap values.
Program:
#include <stdio.h>
 
void swapValue(int a, int b)
{
    int temp = a;
    a = b;
    b = temp;
 
    printf("Inside call by value: %d %d\\n", a, b);
}
 
void swapAddress(int *a, int *b)
{
    int temp = *a;
    *a = *b;
    *b = temp;
}
 
int main()
{
    int a = 10, b = 20;
 
    printf("Original values: %d %d\\n", a, b);
 
    swapValue(a, b);
    printf("After call by value: %d %d\\n", a, b);
 
    swapAddress(&a, &b);
    printf("After call by address: %d %d", a, b);
 
    return 0;
}

18. Write a program to sort an array of n elements using bubble sort.
Program:
#include <stdio.h>

{
    int a[100], n, i, j, temp;
 
    printf("Enter array size: ");
    scanf("%d", &n);
 
    printf("Enter array elements:\\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);
 
    for (i = 0; i < n - 1; i++)
    {
        for (j = 0; j < n - i - 1; j++)
        {
            if (a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    for (i = 0; i < n; i++)
        printf("%d ", a[i]);
 
    return 0;
}

#include <stdio.h>
 
int main()
{

    int r1, c1, r2, c2, i, j, k;
 
    printf("Enter rows and columns of first matrix: ");
    scanf("%d %d", &r1, &c1);
 
    printf("Enter rows and columns of second matrix: ");
    scanf("%d %d", &r2, &c2);
 
    if (c1 != r2)
    {
        printf("Matrix multiplication is not possible");
        return 0;
    }
 
    printf("Enter first matrix:\\n");
    for (i = 0; i < r1; i++)
        for (j = 0; j < c1; j++)
            scanf("%d", &a[i][j]);
 
    printf("Enter second matrix:\\n");
    for (i = 0; i < r2; i++)
        for (j = 0; j < c2; j++)
            scanf("%d", &b[i][j]);
 
    for (i = 0; i < r1; i++)
    {
        for (j = 0; j < c2; j++)
        {
            c[i][j] = 0;
            for (k = 0; k < c1; k++)
                c[i][j] = c[i][j] + a[i][k] * b[k][j];
        }
    }
 
    printf("Product matrix:\\n");
    for (i = 0; i < r1; i++)
    {
        for (j = 0; j < c2; j++)
            printf("%d ", c[i][j]);
        printf("\\n");
    }
 
    return 0;
}
